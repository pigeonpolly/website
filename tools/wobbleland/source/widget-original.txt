const mapFrame = document.getElementById('mapFrame');
const chipRow = document.getElementById('chipRow');
const panel = document.getElementById('panel');
const scrim = document.getElementById('scrim');
const panelBody = document.getElementById('panelBody');
const panelImg = document.getElementById('panelImg');
const panelImgCap = document.getElementById('panelImgCap');

CITIES.forEach(c=>{
  const pin = document.createElement('button');
  pin.className = 'pin';
  pin.style.left = c.coords.left + '%';
  pin.style.top = c.coords.top + '%';
  pin.setAttribute('aria-label', 'Open ' + c.name);
  pin.dataset.key = c.key;
  pin.innerHTML = '<span class="plabel">'+c.name+'</span><span class="dot"></span>';
  pin.addEventListener('click', ()=>openCity(c.key));
  mapFrame.appendChild(pin);

  const chip = document.createElement('button');
  chip.className = 'chip';
  chip.dataset.key = c.key;
  chip.innerHTML = '<img class="crest-icon" src="'+CRESTS[c.key]+'" alt=""><span class="n">'+c.no+'</span>'+c.name;
  chip.addEventListener('click', ()=>openCity(c.key));
  chipRow.appendChild(chip);
});

function sectionHoliday(h){
  return '<div class="holiday"><div class="date">'+h.date+'</div><div><p class="htitle">'+h.title+'</p><p class="hdesc">'+h.desc+'</p></div></div>';
}
function sectionDish(d){
  return '<div class="dish"><p class="dname">'+d.name+'</p><p class="ddesc">'+d.desc+'</p></div>';
}
function sectionStreet(s){
  return '<li><div class="sname">'+s.name+'</div>'+(s.desc && s.desc !== '—' ? '<div class="sdesc">'+s.desc+'</div>' : '')+'</li>';
}
function sectionResident(r){
  let addr = '';
  if(r.home) addr += '<b>Home:</b> '+r.home+'. ';
  if(r.work) addr += '<b>Work:</b> '+r.work+'. ';
  if(r.note) addr += r.note;
  return '<li><div class="rname">'+r.name+'</div><div class="raddr">'+addr+'</div></li>';
}

function openCity(key){
  const c = CITIES.find(x=>x.key===key);
  if(!c) return;

  document.querySelectorAll('.pin').forEach(p=>p.classList.toggle('active', p.dataset.key===key));
  document.querySelectorAll('.chip').forEach(p=>p.classList.toggle('active', p.dataset.key===key));

  panelImg.src = IMAGES[c.key];
  panelImg.alt = c.name + ' — detail from the Wobbleland map';
  panelImgCap.textContent = 'Entry No. ' + c.no + (c.mapLabel ? ' · labeled "'+c.mapLabel+'"' : '');

  let html = '';
  html += '<div class="title-row">';
  html += '<div class="crest-badge"><img src="'+CRESTS[c.key]+'" alt="Crest of '+c.name+'"></div>';
  html += '<div class="titles">';
  html += '<div class="entry-no">Bestiary Entry No. '+c.no+'</div>';
  html += '<h2>'+c.name+'</h2>';
  html += '</div></div>';
  html += '<div class="motto">'+c.motto+'</div>';
  html += '<p class="desc">'+c.desc+'</p>';

  if(c.holidays.length){
    html += '<div class="section-title">Local holidays</div>';
    html += c.holidays.map(sectionHoliday).join('');
  }
  if(c.dishes.length){
    html += '<div class="section-title">Signature dishes</div>';
    html += c.dishes.map(sectionDish).join('');
  }
  if(c.streets.length){
    html += '<div class="section-title">Streets &amp; landmarks</div>';
    html += '<ul class="streets">' + c.streets.map(sectionStreet).join('') + '</ul>';
  }
  if(c.residents.length){
    html += '<div class="section-title">Notable residents</div>';
    html += '<ul class="residents">' + c.residents.map(sectionResident).join('') + '</ul>';
  }
  html += '<div class="neighbors">'+c.neighbors+'</div>';

  panelBody.innerHTML = html;
  panel.classList.add('open');
  panel.setAttribute('aria-hidden','false');
  scrim.classList.add('open');
  panelBody.parentElement.scrollTop = 0;
}

function closePanel(){
  panel.classList.remove('open');
  panel.setAttribute('aria-hidden','true');
  scrim.classList.remove('open');
  document.querySelectorAll('.pin').forEach(p=>p.classList.remove('active'));
  document.querySelectorAll('.chip').forEach(p=>p.classList.remove('active'));
}

document.getElementById('panelClose').addEventListener('click', closePanel);
scrim.addEventListener('click', closePanel);
document.addEventListener('keydown', e=>{ if(e.key==='Escape') closePanel(); });
</script>
